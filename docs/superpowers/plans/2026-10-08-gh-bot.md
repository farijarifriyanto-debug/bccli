# `bccli gh-bot` Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** webhook GitHub loopback off-by-default: HMAC atas byte mentah, admission
atomik (duplikat → kapasitas → reservasi), filter `/bccli` allowlist, agent tanpa tool
dengan SATU deadline melewati collect+draft+post (signal ke subprocess `gh`), post via
stdin `gh api --input -`, 503=drop + pemulihan manual (tanpa retry otomatis).

**Architecture:** 4 modul injected (`ghBotWebhook` verifikasi byte-decode-parse,
`ghBotDispatch` hub state-machine inFlight+LRU+close, `ghBotRunner` konteks `--repo`
+draft+post stdin, `ghBotServe` HTTP status-table + glue nyata) + wiring CLI/HELP/i18n.
Semua eksternal (spawn gh, runtime agent, jam, secret accessor, out/err) di-inject.

**Tech Stack:** Node ≥22, TypeScript strict, `node:crypto` HMAC, `node:http`,
`spawn` args-array `shell:false` + `signal`, vitest 5, tanpa dependency baru.

**Spec:** `docs/superpowers/specs/2026-10-08-gh-bot-design.md`

## Global Constraints

- **DOKUMEN ini BELUM di-commit** — commit docs hanya setelah review lolos; setelah itu
  TDD per task + commit lokal per task.
- **TANPA GitHub nyata**: tanpa webhook/tunnel/komentar nyata; semua test fake.
- **Tanpa retry otomatis** di bccli (503 = drop; pemulihan = redelivery manual GitHub UI).
- **Tanpa `spawnSync` utk gh** (revisi glue #1): semua spawn gh ASINKRON (`spawn`,
  `shell:false`, `signal`, stdin pipe + handler EPIPE/error/close settle-sekali) —
  diuji dgn subprocess lokal NYATA (bukan fake signal saja).
- Agent SELALU `setTools([])`; teks tak tepercaya tak pernah masuk shell (argv array /
  stdin); secret & payload tak pernah di-log; batas ukuran dalam **byte UTF-8 tanpa
  memotong karakter**.
- Assertion teks `t()` EN → pin `setLanguage('en')` + `afterAll('id')`; suite bahasa ID.
- Gates/task: tsc 0, lint 0/0, `npm test` hijau. Edit file via tool edit/write (tanpa
  char kontrol mentah di payload — gunakan token ASCII jika perlu).

  GITHUB_HEADERS,

  'x-hub-delivery': 'd-1',
  'x-hub-event': 'issue_comment',
  ...over,
})

  method: 'POST',
  contentType: 'application/json',
  maxBodyBytes: 1_048_576,
  getSecret: () => SECRET,
  ...over,

test('valid → proyeksi lengkap termasuk isPR dari payload', () => {
  const r = verifyAndParse(body, H(), OPTS())
  expect(r.ok).toBe(true)
  if (r.ok) expect(r.delivery).toMatchObject({ delivery: 'd-1', repo: 'acme/demo', number: 42, author: 'alice', authorType: 'User', isPR: false })
  // payload PR: field pull_request melekat di issue_comment
  const pr = { ...payload, pull_request: { url: 'x' } }
  const prBuf = Buffer.from(JSON.stringify(pr))
  const r2 = verifyAndParse(prBuf, H({}, prBuf), OPTS())
  expect(r2.ok).toBe(true)
  if (r2.ok) expect(r2.delivery.isPR).toBe(true)
})

test('urutan: method → content-type → size → header → secret → HMAC(BUFFER) → decode fatal → parse', () => {
  expect(verifyAndParse(body, H(), OPTS({ method: 'GET' }))).toMatchObject({ status: 405 })
  expect(verifyAndParse(body, H(), OPTS({ contentType: 'text/plain' }))).toMatchObject({ status: 415 })
  expect(verifyAndParse(Buffer.alloc(50), H({}, Buffer.alloc(50)), OPTS({ maxBodyBytes: 10 }))).toMatchObject({ status: 413 })
  expect(verifyAndParse(body, H({ 'x-hub-delivery': undefined }), OPTS())).toMatchObject({ status: 400 })
  expect(verifyAndParse(body, H(), OPTS({ getSecret: () => undefined }))).toMatchObject({ status: 503 })
  // signature invalid → 401 SEBELUM parse
  const spy = vi.fn((t: string) => JSON.parse(t))
  expect(verifyAndParse(body, H({ 'x-hub-signature-256': 'sha256=dead' }), OPTS(), spy)).toMatchObject({ status: 401 })
  expect(spy).not.toHaveBeenCalled()
  // byte non-UTF-8 dengan HMAC VALID → 400 (decode fatal), parse tak terpanggil
  const invalidUtf8 = Buffer.from([0xff, 0xfe, 0x41])
  const spy2 = vi.fn((t: string) => JSON.parse(t))
  expect(verifyAndParse(invalidUtf8, H({}, invalidUtf8), OPTS(), spy2)).toMatchObject({ status: 400 })
  expect(spy2).not.toHaveBeenCalled()
})

test('JSON corrupt & array ditandatangani ATAS BODY ITU → HMAC lolos, 400 dari parse (revisi #7)', () => {
  const corrupt = Buffer.from('{corrupt', 'utf8')
  const spyC = vi.fn((t: string) => JSON.parse(t))
  expect(verifyAndParse(corrupt, H({}, corrupt), OPTS(), spyC)).toMatchObject({ status: 400 })
  expect(spyC).toHaveBeenCalledTimes(1) // benar-benar menguji parse
  const arr = Buffer.from('["array"]', 'utf8')
  const spyA = vi.fn((t: string) => JSON.parse(t))
  expect(verifyAndParse(arr, H({}, arr), OPTS(), spyA)).toMatchObject({ status: 400 })
  expect(spyA).toHaveBeenCalledTimes(1)
  // field event hilang pada objek valid-BER-SIGNATURE → 400
  const bare = Buffer.from(JSON.stringify({ action: 'created' }))
  expect(verifyAndParse(bare, H({}, bare), OPTS())).toMatchObject({ status: 400 })
})

test('rotasi secret via accessor (revisi #6): secret lama → 401; secret baru → lolos', () => {
  let secret = SECRET
  const opts = OPTS({ getSecret: () => secret })
  expect(verifyAndParse(body, H(), opts).ok).toBe(true)
  secret = 'rotated-secret'
  expect(verifyAndParse(body, H(), opts)).toMatchObject({ status: 401 })
  const reSigned = Buffer.concat([body]) // body sama
  expect(verifyAndParse(reSigned, { ...H(), 'x-hub-signature-256': signBody(reSigned, 'rotated-secret') }, opts).ok).toBe(true)
})

test('signBody deterministik format sha256=<hex>; rawBody byte apa adanya', () => {
  expect(signBody(body, SECRET)).toMatch(/^sha256=[0-9a-f]{64}$/)
  const messy = Buffer.from('{\n  "a": 1  }', 'utf8')
  expect(signBody(messy, SECRET)).toBe('sha256=' + createHmac('sha256', SECRET).update(messy).digest('hex'))
})

test('extractCommand: valid/whitespace/kosong/bukan perintah/batas kata', () => {
  expect(extractCommand('/bccli fix the bug')).toBe('fix the bug')
  expect(extractCommand('  /bccli   hello  ')).toBe('hello')
  expect(extractCommand('/bccli')).toBeNull()
  expect(extractCommand('/bcl')).toBeNull()
  expect(extractCommand('/bcclix nope')).toBeNull()
  expect(extractCommand('plain comment')).toBeNull()
})

test('truncateUtf8Bytes: tak pernah memotong karakter — emoji/CJK utuh, byteLength ≤ max', () => {
  const text = '😀漢字'.repeat(500)
  const out = truncateUtf8Bytes(text, 100)
  expect(Buffer.byteLength(out, 'utf8')).toBeLessThanOrEqual(100)
  expect([...out].every((ch) => !ch.includes('�'))).toBe(true) // tanpa replacement char
  // pasangan surrogate tak pernah terbelah
  expect(out).not.toMatch(/[\uD800-\uDBFF]$/)
  expect(truncateUtf8Bytes('pendek', 100)).toBe('pendek')
})

test('truncateHeadTailUtf8: ≤max byte, 60/40 kepala-ekor + marker, karakter utuh', () => {
  const head = 'H'.repeat(600)
  const tail = '😀'.repeat(300) // 4 byte/char
  const out = truncateHeadTailUtf8(head + tail, 200)
  expect(Buffer.byteLength(out, 'utf8')).toBeLessThanOrEqual(200)
  expect(out).toContain('truncated')
  expect(out.startsWith('H')).toBe(true)
  expect(out.endsWith('😀') || out.endsWith('…')).toBe(true)
  expect(truncateHeadTailUtf8('small', 100)).toBe('small')
})

import { setLanguage } from '../src/i18n'
import { GhBotHub, ghBotConfigFrom } from '../src/ghBotDispatch'
import type { CommentDelivery } from '../src/ghBotWebhook'

beforeAll(() => setLanguage('en'))
afterAll(() => setLanguage('id'))

const cfg = (over: Record<string, unknown> = {}) =>
  ghBotConfigFrom({ enabled: true, repos: ['acme/demo'], users: ['alice'], ...over })
const d = (over: Partial<CommentDelivery> = {}): CommentDelivery => ({
  delivery: 'd-' + Math.random().toString(36).slice(2), event: 'issue_comment', action: 'created',
  repo: 'acme/demo', number: 1, author: 'alice', authorType: 'User', body: '/bccli hi', ...over,
})
function makeHub(c = cfg(), over: Partial<import('../src/ghBotDispatch').GhBotHubDeps> = {}) {
  const gate: { release: () => void }[] = []
  const calls = { jobs: [] as string[], signals: [] as AbortSignal[] }
  const deps = {
    out: () => {}, err: () => {},
    runJob: (del: CommentDelivery, signal: AbortSignal) => {
      calls.jobs.push(del.delivery)
      calls.signals.push(signal)
      return new Promise<{ ok: boolean; reason?: string }>((resolve) => {
        gate.push(() => resolve({ ok: true }))
        signal.addEventListener('abort', () => resolve({ ok: false, reason: 'aborted' }))
      })
    },
    ...over,
  }
  const h = new GhBotHub(c, deps)
  return { h, calls, releaseAll: () => { while (gate.length) gate.shift()?.() } }
}

1. **Filter** (tanpa state): `action:'edited'|'deleted'`, `event` lain, `authorType:'Bot'`,

   → `'filtered'`; `stats().inflight === 0` untuk semua.
2. **Admission atomik**: job sukses (`releaseAll` + await whenIdle) → LRU `completed:1`;
   ulang id sama → `'dedup'` (jobs=1). Delivery aktif (gate tertahan) → ulang →
   `'inflight'` (jobs=1) — **duplicate aktif saat LRU penuh**: `dedupCap` kecil (mis. 2)
   diisi delivery lain + delivery aktif diulang → tetap `'inflight'`, jobs tidak nambah.
3. **Redelivery setelah queue penuh (revisi #2)**: `maxQueue:1`, job pertama aktif
   (gate) → kedua `'queued'` (antrian=1) → ketiga `'full'` + **`stats().completed` dan
   dedup tidak berubah** → release semua → await idle → **delivery ketiga yang sama
   kembali → `'queued'`** (full TIDAK menyimpan dedup) → release → sukses.
4. **Job gagal → bisa diulang (revisi #1/#4)**: `runJob` resolve `{ok:false}` →

   `'dedup'` berikutnya.
5. **Deadline signal tunggal**: signal yang diteruskan ke runJob = `AbortSignal.any`
   (cek: setelah `timeoutMs: 10` fake clock/timers nyata → signal aborted otomatis;
   gunakan `timeoutMs: 15` + await job gagal `reason:'aborted'|'timeout'`).
   (Detail fake-timers: pakai timeoutMs kecil + await promise job.)
6. **Close/disposal (revisi #3 + review lifecycle + rapihan #4)**: job aktif + antrian
   tertahan → `const summary = await h.close()` → aktif ter-abort (signal aborted →
   runJob resolve ok:false dgn reason `aborted`), antrian dibuang (`jobs` tidak pernah
   memanggil job antrian) DAN **ID job antrian itu dihapus dari `inFlight`** →
   **`stats()` = `{queued:0, active:0, inflight:0}`** (assert eksplisit setelah close),
   `whenIdle` resolve, `accept` berikutnya `'closed'`; **ringkasan**:
   `summary = { aborted: ≥1, terminationConfirmed: ≥1, terminationUnconfirmed: 0 }`
   (jalur abort biasa = confirmed) dan `err` log TIDAK memuat `UNCONFIRMED`.
   **6b (dua arah rapihan #4 + koreksi #1)**: `over.runJob` custom yang, saat signal closed abort,
   resolve `{ok:false, reason:'termination-unconfirmed'}` → `summary.terminationUnconfirmed === 1`
   + `summary.terminationConfirmed === 0` + **`stats().completed === 0` (TIDAK masuk
   completed LRU)** + `errLines` memuat `UNCONFIRMED 1 (child may still be running)` —
   membuktikan kondisi hard-timeout diteruskan ke pelaporan close tanpa klaim child mati.
7. **Config**: `ghBotConfigFrom` tipe salah (`repos:'x'`) → throw; default sesuai §2.
8. **Cleanup — DUA tes terpisah (revisi glue #4 + review #2; JANGAN mengklaim throw
   sinkron teruji lewat Promise.reject)**:
   - **8a throw SINKRON**: `over: { runJob: () => { throw new Error('sync-boom') } }`
     (fungsi melempar langsung, tidak mengembalikan promise) → setelah settle:
     `stats().active === 0`, `stats().inflight === 0`, **antrean job ke-2 tetap
     terpump** (pakai harness dgn overload berselang: call-1 throw sinkron, call-2
     resolve gate), `whenIdle()` resolve, delivery job pertama bisa diulang
     (`'queued'` lagi — tidak masuk LRU);

    import { spawn } from 'node:child_process'
    export function createGhSpawn(
      cmd = 'gh',
      o?: { spawnImpl?: typeof spawn; killGraceMs?: number; hardMs?: number },
    ): GhDeps['spawn'] {
      const spawnImpl = o?.spawnImpl ?? spawn
      const killGraceMs = o?.killGraceMs ?? 2000
      const hardMs = o?.hardMs ?? killGraceMs + 3000
      return (argv, opts) =>
        new Promise((resolve) => {
          const child = spawnImpl(cmd, argv, { shell: false, stdio: ['pipe', 'pipe', 'pipe'], signal: opts?.signal })
          let stdout = ''
          let stderr = ''
          let lastError = ''
          let settled = false
          let graceTimer: ReturnType<typeof setTimeout> | null = null
          let hardTimer: ReturnType<typeof setTimeout> | null = null
          const clearTimers = (): void => {
            if (graceTimer) clearTimeout(graceTimer)
            if (hardTimer) clearTimeout(hardTimer)
            graceTimer = null
            hardTimer = null
          }
          // Rapihan #3: handler abort BERNAMA — di-remove di settle() pada SEMUA jalur
          // (close normal maupun hard timeout); removeEventListener no-op bila sudah once-fire.
          const onAbort = (): void => startBoundedTermination()
          // Koreksi review #2: settle HANYA dari close KECUALI hard-timeout; settle
          // MELEPAS semua listener milik runner (data stdin/stdout/stderr, close/error
          // anak) dan menghentikan pengumpulan — data telat tak menambah buffer.
          const onStdout = (c: string): void => {
            stdout += c
          }
          const onStderr = (c: string): void => {
            stderr += c
          }
          const onChildError = (e: Error): void => {
            lastError = e.message
            startBoundedTermination()
          }
          const onChildClose = (code: number | null): void => settle({ code: code ?? 1, stdout, stderr: stderr || lastError }, false)
          const settle = (result: { code: number; stdout: string; stderr: string }, hard: boolean): void => {
            if (settled) return
            settled = true // settle-sekali (event telat diabaikan — tanpa unhandled)
            clearTimers()
            opts?.signal?.removeEventListener('abort', onAbort)
            child.stdout.off('data', onStdout)
            child.stderr.off('data', onStderr)
            child.stdout.off('error', noop)
            child.stderr.off('error', noop)
            child.stdin.off('error', onStdinError)
            child.off('error', onChildError)
            child.off('close', onChildClose)
            if (hard) {
              // Jalur hard-timeout: bersihkan pipe lokal (destroy) — termination BELUM
              // terkonfirmasi (jangan mengklaim child mati).
              for (const s of [child.stdin, child.stdout, child.stderr]) {
                try {
                  s.destroy()
                } catch {
                  /* stream sudah tertutup */
                }
              }
            }
            resolve(result)
          }
          // Rapihan #1/#2: eskalasi EKSPLISIT SIGKILL + marker SELALU ada.
          // POSIX: SIGKILL tak ditangani child; Windows: kill('SIGKILL') → terminasi
          // paksa (TerminateProcess) — sinyal non-INT/TERM di Windows selalu memaksa.
          const startBoundedTermination = (): void => {
            if (settled || graceTimer) return
            graceTimer = setTimeout(() => {
              if (!settled) {
                try {
                  child.kill('SIGKILL') // argumen WAJIB 'SIGKILL' — diuji oleh tes (f)
                } catch {
                  /* child sudah mati */
                }
              }
            }, killGraceMs)
            hardTimer = setTimeout(() => {
              if (!settled) {
                // Marker SELALU ikut — walau stderr/lastError sudah terisi (tes (f)).
                const marker = '<child did not close after termination>'
                settle(
                  {
                    code: 1,
                    stdout,
                    stderr: [stderr, lastError, marker].filter(Boolean).join('\n'),
                  },
                  true, // hard → destroy pipe + cleanup listener
                )
              }
            }, hardMs)
          }
          const noop = (): void => {}
          const onStdinError = (e: NodeJS.ErrnoException): void => {
            if (e.code !== 'EPIPE') stderr += `\nstdin error: ${e.message}` // EPIPE: child keluar duluan — abaikan
          }
          child.stdout.setEncoding('utf8').on('data', onStdout)
          child.stderr.setEncoding('utf8').on('data', onStderr)
          // Rapihan #3: handler error noop — event telat setelah settle tidak menjatuhkan proses.
          child.stdout.on('error', noop)
          child.stderr.on('error', noop)
          child.stdin.on('error', onStdinError)
          child.stdin.end(opts?.stdin ?? '')
          // Review #4: 'error' HANYA mencatat + memicu watchdog — promise TIDAK pernah
          // resolve sebelum 'close' (KECUALI jalur hard-timeout force-settle, di atas).
          // ENOENT tetap resolve via close.
          child.on('error', onChildError)
          if (opts?.signal) {
            opts.signal.addEventListener('abort', onAbort)
            if (opts.signal.aborted) onAbort()
          }
          child.on('close', onChildClose)
        })
    }

function makeRunJob(
